# CONFIGURATION

# Allow all crawlers
User-agent: *

# Allow crawling of all public content
Allow: /

# ─────────────────────────────────────────────────────────────────────────────
# SITEMAP LOCATION
# ⚠️ Update this URL when you deploy to a new domain
# ─────────────────────────────────────────────────────────────────────────────
Sitemap: https://www.eneanushi.space/sitemap.xml

# ─────────────────────────────────────────────────────────────────────────────
# CRAWL RATE (optional)

# Remove if you want faster crawling
# ─────────────────────────────────────────────────────────────────────────────
Crawl-delay: 1

# ─────────────────────────────────────────────────────────────────────────────
# BLOCKED PATHS
# Add paths you don't want search engines to index
# ─────────────────────────────────────────────────────────────────────────────

# Block API routes (if any)
Disallow: /api/

# Block raw JSON files
Disallow: /*.json$

# Block source code directory (Vite serves from here in dev)
Disallow: /src/

# Block hidden files
Disallow: /.*

# ─────────────────────────────────────────────────────────────────────────────
# ALLOWED ASSET PATHS
# Explicitly allow crawlers to index your assets
# ─────────────────────────────────────────────────────────────────────────────

# Favicon and Open Graph images
Allow: /favicon/

# Texture and image assets
Allow: /textures/

# 3D model assets (for SEO purposes, though not directly indexable)
Allow: /models/

# Document files (like PDF resume)
Allow: /documents/

# ─────────────────────────────────────────────────────────────────────────────
# HOST DIRECTIVE
# ⚠️ Update this URL when you deploy to a new domain
# Tells crawlers the preferred domain (www vs non-www)
# ─────────────────────────────────────────────────────────────────────────────
Host: https://www.eneanushi.space
